library(jsonlite)
row <- function(term, estimate, se=NULL, n_obs) {
  list(term=term, estimate=unname(as.numeric(estimate)),
       se=if (is.null(se)) NULL else unname(as.numeric(se)), n_obs=as.integer(n_obs))
}
report <- function(id, rows) {
  write_json(list(task_id=id, estimates=rows), "example_result.json",
             auto_unbox=TRUE, null="null", digits=16, pretty=TRUE)
}
# Separate teaching data. Time values are one-based observation labels.
data <- data.frame(time=c(3,1,7,5,9,2,6,8,4),visits=c(24,21,30,27,33,18,23,25,20))
data <- data[order(data$time),]
window_levels <- 6
forecast_times <- c(7,8,9)
forecasts <- numeric(length(forecast_times))
for (i in seq_along(forecast_times)) {
  f <- forecast_times[i]
  history <- data[data$time>=f-window_levels & data$time<=f-1,]
  stopifnot(identical(as.numeric(history$time),as.numeric(seq(f-window_levels,f-1))))
  levels <- history$visits
  # Six levels give five pairs. At f=7 these are (1,2),...,(5,6).
  lagged <- head(levels,-1)
  current <- tail(levels,-1)
  fit <- lm(current ~ lagged)
  stopifnot(length(current)==window_levels-1)
  # Forecast f from observed visits at f-1. Exclude observed visits at f.
  forecasts[i] <- coef(fit)[1]+coef(fit)[2]*tail(levels,1)
}
# Refit each date. Time8 uses observed time7, not forecast7.
actual <- data$visits[match(forecast_times,data$time)]
rmse <- sqrt(mean((actual-forecasts)^2))
rows <- list(row("clinic_rmse",rmse,NULL,length(forecast_times)))
for (i in seq_along(forecast_times)) {
  rows[[length(rows)+1]] <- row(paste0("clinic_forecast_",forecast_times[i]),forecasts[i],NULL,window_levels-1)
}
report("clinic_three_rolling_forecasts",rows)
