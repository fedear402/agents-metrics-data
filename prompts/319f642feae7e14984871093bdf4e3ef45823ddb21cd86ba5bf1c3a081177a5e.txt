```python
import json
import numpy as np

def report(**values):
    rows = [
        {'term': 'speed_slope', 'estimate': float(values['slope']), 'se': float(values['slope_se']), 'n_obs': int(values['people'])},
        {'term': 'high_at_zero', 'estimate': float(values['high_at_zero']), 'se': None, 'n_obs': int(values['people'])},
    ]
    output = {'task_id': 'ordered_logit_separate_example', 'estimates': rows}
    with open('example_result.json', 'w') as stream:
        json.dump(output, stream, indent=2, allow_nan=False)

from scipy.special import expit
from statsmodels.miscmodels.ordinal_model import OrderedModel
rng = np.random.default_rng(1103)
speed = rng.normal(size=90)
latent = .7*speed + rng.logistic(size=90)
category = np.digitize(latent,[-.5,.9])+1
model = OrderedModel(category,speed[:,None],distr='logit')
fit = model.fit(method='bfgs',disp=False,gtol=1e-8)
threshold = model.transform_threshold_params(fit.params)
probability = model.predict(fit.params,exog=np.array([[0.]]))[0,2]
assert np.isclose(probability,1-expit(threshold[2])) and threshold[1]<threshold[2]
report(slope=fit.params[0], slope_se=fit.bse[0], high_at_zero=probability, people=90)
```
