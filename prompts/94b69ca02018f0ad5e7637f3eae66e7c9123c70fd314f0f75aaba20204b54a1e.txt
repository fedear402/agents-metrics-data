```python
import itertools
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize_scalar
# Three separate groups have one observed month and two simultaneous exits.
x = np.array([0,0,0,1,1,1])
observed_sets = [(0,3),(3,4),(1,4)]
all_sets = list(itertools.combinations(range(6),2))
sums = np.array([x[list(s)].sum() for s in all_sets])
observed_sums = np.array([x[list(s)].sum() for s in observed_sets])
def loglike(beta):
    return float(np.sum(beta*observed_sums - logsumexp(beta*sums)))
fit = minimize_scalar(lambda beta:-loglike(beta),bounds=(-3,3),method='bounded')
beta = float(fit.x)
log_probability = beta*observed_sums[0] - logsumexp(beta*sums)
print({'training_coefficient':beta,'conditional_odds_ratio':float(np.exp(beta)), 'log_likelihood':loglike(beta), 'first_event_set_probability':float(np.exp(log_probability)), 'first_event_set_log_probability':float(log_probability)})
```
