A smaller example uses separate synthetic data to illustrate the method.
```stata
clear all
set more off
set type double
input group experience wage
0 1 11
0 2 13
0 4 14
0 6 18
0 7 19
1 2 14
1 3 17
1 5 21
1 7 22
1 9 28
end
foreach g in 0 1 {
    quietly regress wage experience if group==`g'
    scalar b`g'c=_b[_cons]
    scalar b`g'x=_b[experience]
    quietly summarize experience if group==`g',meanonly
    scalar m`g'x=r(mean)
    quietly summarize wage if group==`g',meanonly
    scalar m`g'y=r(mean)
}
scalar wage_gap=m1y-m0y
scalar explained_gap=(m1x-m0x)*b0x
scalar unexplained_gap=(b1c-b0c)+m1x*(b1x-b0x)
assert abs(wage_gap-explained_gap-unexplained_gap)<1e-10
foreach term in wage_gap explained_gap unexplained_gap {
    display "ANSWER: `term' estimate=" %24.16g `term' " se=NA n_obs=10"
}
exit, clear
```
