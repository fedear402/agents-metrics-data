fit <- survival::survreg(survival::Surv(reading, reading > 1, type="left") ~ temperature,
                         data=d, dist="gaussian", control=survival::survreg.control(rel.tolerance=1e-12))
# survreg supplies inverse observed information for beta and log(scale).
V <- vcov(fit)
stopifnot(min(eigen(V, symmetric=TRUE, only.values=TRUE)$values) > 0)
rows <- list(
  list(term="temperature_slope", estimate=unname(coef(fit)["temperature"]), se=sqrt(V["temperature","temperature"]), n_obs=100L),
  list(term="latent_scale", estimate=unname(fit$scale), se=NULL, n_obs=100L))
