A simpler example uses separate data and illustrates the method.
```python
import numpy as np
import pandas as pd
from linearmodels.iv import IV2SLS
# Local projections use a shared set of calendar origins in a smaller panel.
rng=np.random.default_rng(433);rows=[]
for unit in range(4):
    for t in range(10):
        z=rng.normal();u=rng.normal();x=z+0.4*u
        rows.append((unit,t,x,z,unit+0.2*t+1.3*x+u))
d=pd.DataFrame(rows,columns=['unit','period','x','z','y']);d=d[~((d.unit==1)&(d.period==4))]
sample=d.copy()
for h in [-1,1]:
    q=d[['unit','period','y']].copy();q.period-=h
    sample=sample.merge(q.rename(columns={'y':f'y{h}'}),on=['unit','period'])
C=pd.concat([pd.Series(1.,index=sample.index,name='intercept'),sample['y-1'].rename('lag_y'),pd.get_dummies(sample.unit,prefix='unit',drop_first=True,dtype=float),pd.get_dummies(sample.period,prefix='period',drop_first=True,dtype=float)],axis=1)
assert C.columns.is_unique
assert np.linalg.matrix_rank(C)==C.shape[1]
print({f'lp_h{h}':float(IV2SLS(sample['y' if h==0 else 'y1'],C,sample.x,sample.z).fit().params['x']) for h in [0,1]})
```
