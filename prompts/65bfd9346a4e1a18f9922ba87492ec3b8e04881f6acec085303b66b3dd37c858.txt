iv <- function(y,C,x,z,w=rep(1,length(y))) {
  X <- cbind(C,x)*sqrt(w); Z <- cbind(C,z)*sqrt(w)
  fitted <- qr.fitted(qr(Z),X)
  as.numeric(qr.coef(qr(fitted),y*sqrt(w)))[ncol(X)]
}
d <- files[['example.csv']]
base <- as.matrix(d[c('base1','base2')]); early <- as.matrix(d[c('early1','early2')])
# Divide nominal dollars by the price index, then convert to thousands per decade.
d$x <- as.vector(base %*% (c(2e6,5e6)/colSums(base)/(120/100)/1000*(10/5)))/d$base_total
d$z <- as.vector(early %*% (c(3e6,4e6)/colSums(early)/(120/100)/1000*(10/5)))/d$early_total
out <- c(first_exposure=d$x[1],first_instrument=d$z[1],iv_slope=iv(d$outcome,matrix(1,nrow(d),1),d$x,d$z,d$weight))
cat(jsonlite::toJSON(as.list(out),auto_unbox=TRUE,digits=12),'\n')
