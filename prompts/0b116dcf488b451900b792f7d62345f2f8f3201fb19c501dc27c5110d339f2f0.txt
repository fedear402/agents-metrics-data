A service starts at time 3 for two towns. Other towns adopt later or never. Use times 2, 3, and 4. Controls must remain untreated through time 4. Keep only towns observed at all three window times. Estimate the treated-minus-control change from time 2 to the mean of times 3 and 4.

Determine control eligibility from the end of this comparison window. Check balance inside this window. A town treated at its last period is ineligible. An otherwise eligible town with a missing window observation is also removed.

The target repeats this process for three cohorts and aggregates with retained-treated counts.

This separate example writes example_result.json. The task reads input.csv and reports its own results.

```python
import json
import numpy as np

def report(**values):
    rows = [
        {'term': 'town_effect', 'estimate': float(values['effect']), 'se': None, 'n_obs': int(values['rows'])},
        {'term': 'treated_towns', 'estimate': float(values['retained_treated']), 'se': None, 'n_obs': int(values['rows'])},
    ]
    output = {'task_id': 'cohort_stack_separate_example', 'estimates': rows}
    with open('example_result.json', 'w') as stream:
        json.dump(output, stream, indent=2, allow_nan=False)

town = np.repeat(np.arange(1,7),3)
time = np.tile([2,3,4],6)
adoption = np.repeat([3,3,0,5,4,0],3)
level = np.repeat([10.,14.,8.,12.,9.,15.],3)
y = level + .5*time + 2*((adoption==3)&(time>=3))
y += np.array([0,.2,-.1,.1,-.2,.3,0,.1,-.1,.1,0,.2,0,.5,.8,.1,.3,-.2])
observed = ~((town==6)&(time==3))
eligible = (adoption==3)|(adoption==0)|(adoption>4)
keep_towns = [unit for unit in np.unique(town) if np.sum(observed&eligible&(town==unit))==3]
changes = {}
for unit in keep_towns:
    rows = town==unit
    changes[unit] = y[rows&(time>=3)].mean()-y[rows&(time==2)][0]
treated_units = [u for u in keep_towns if adoption[town==u][0] == 3]
control_units = [u for u in keep_towns if u not in treated_units]
effect = np.mean([changes[u] for u in treated_units])-np.mean([changes[u] for u in control_units])
assert keep_towns==[1,2,3,4]
import pandas as pd
keep = observed & np.isin(town,keep_towns)
did = ((adoption[keep]==3)&(time[keep]>=3)).astype(float)
# An intercept accompanies one omitted category in each fixed-effect set.
design = np.column_stack([np.ones(keep.sum()),did,
    pd.get_dummies(town[keep],drop_first=True,dtype=float),
    pd.get_dummies(time[keep],drop_first=True,dtype=float)])
assert np.linalg.matrix_rank(design)==design.shape[1]
fe_effect = np.linalg.lstsq(design,y[keep],rcond=None)[0][1]
assert np.isclose(fe_effect,effect,atol=1e-10)
report(effect=effect, retained_treated=len(treated_units), rows=3*len(keep_towns))
```
