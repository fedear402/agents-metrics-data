Explanation of this example: Let q count excluded instruments and df=n-p count residual degrees of freedom in the unrestricted regression. F(beta)=((SSR_R(beta)-SSR_U(beta))/q)/(SSR_U(beta)/df). Acceptance is SSR_R-(1+q*critical/df)*SSR_U<=0. Expand this into A*beta^2+B*beta+C<=0. For two distinct real boundaries, A>0 accepts the interval and A<0 accepts the outer rays. Derive the sign and membership from the example data. Other data can require separate degenerate-set handling.

ry <- residuals(lm(harvest~1,data=d)); rd <- residuals(lm(dose~1,data=d))
full_y <- lm(harvest~encouragement+reminder,data=d)
full_d <- lm(dose~encouragement+reminder,data=d)
uy <- residuals(full_y); ud <- residuals(full_d)
q <- 2L; df <- df.residual(full_y)
critical <- qf(.95,q,df)
# Joint F divides the reduction in SSR by q, not one.
k <- q*critical/df
A <- sum(rd^2)-(1+k)*sum(ud^2)
B <- -2*(sum(ry*rd)-(1+k)*sum(uy*ud))
C <- sum(ry^2)-(1+k)*sum(uy^2)
stopifnot(abs(A)>1e-10,B^2-4*A*C>0)
bounds <- sort(Re(polyroot(c(C,B,A)))) # Ascending polynomial powers in R.
unbounded <- as.integer(A<0)
# A>0 accepts the interval. A<0 accepts the two outer rays.
accepts <- function(beta) A*beta^2+B*beta+C<=0
for (b in bounds) {
  transformed <- d$harvest-b*d$dose
  restricted_fit <- lm(transformed~1,data=d)
  full_fit <- lm(transformed~encouragement+reminder,data=d)
  F <- ((deviance(restricted_fit)-deviance(full_fit))/q)/(deviance(full_fit)/df)
  stopifnot(abs(F-critical)<1e-8)
}
rows <- list(
