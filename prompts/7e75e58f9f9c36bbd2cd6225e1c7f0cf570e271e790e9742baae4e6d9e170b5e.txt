import json
import numpy as np

def report(**values):
    rows = [
        {'term': 'distance_choice2', 'estimate': float(values['distance_cat2']), 'se': float(values['slope_se']), 'n_obs': int(values['people'])},
        {'term': 'choice2_at_zero', 'estimate': float(values['probability_cat2']), 'se': None, 'n_obs': int(values['people'])},
    ]
    output = {'task_id': 'multinomial_logit_separate_example', 'estimates': rows}
    with open('example_result.json', 'w') as stream:
        json.dump(output, stream, indent=2, allow_nan=False)

import statsmodels.api as sm
import pandas as pd
from scipy.special import softmax
rng = np.random.default_rng(1207)
distance = rng.normal(size=120)
probability = softmax(np.column_stack([np.zeros(120),.2+.5*distance,-.3-.7*distance]),axis=1)
choice = np.array([rng.choice(3,p=p) for p in probability])
X = pd.DataFrame({"const":np.ones(len(distance)),"distance":distance})
fit = sm.MNLogit(choice,X).fit(disp=False,tol=1e-10)
# Outcomes are 0,1,2; zero is the base. Parameter columns enumerate NONBASE equations.
outcomes = [0,1,2]
assert np.array_equal(np.unique(choice),outcomes)
nonbase_outcomes = outcomes[1:]
# With pandas input, raw parameter columns 0 and 1 correspond to outcomes 1 and 2.
assert list(fit.params.columns)==[0,1]
coefficients = fit.params.copy()
errors = fit.bse.copy()
coefficients.columns = nonbase_outcomes
errors.columns = nonbase_outcomes
prediction_input = pd.DataFrame({"const":[1.],"distance":[0.]})
predicted = pd.Series(np.asarray(fit.predict(prediction_input))[0],index=outcomes)
manual = softmax(np.r_[0.,coefficients.loc["const",nonbase_outcomes]])
assert np.allclose(predicted.to_numpy(),manual) and np.isclose(predicted.sum(),1)
report(distance_cat2=coefficients.loc["distance",2], slope_se=errors.loc["distance",2],
       probability_cat2=predicted.loc[2], people=120)
