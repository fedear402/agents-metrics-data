# The cutpoints supply location. polr does not add a separate free intercept.
d$category <- ordered(d$category, levels=1:3)
fit <- MASS::polr(category ~ speed, data=d, method="logistic", Hess=TRUE,
                  control=list(reltol=1e-12, maxit=1000))
V <- vcov(fit)
# polr reports actual ordered cutpoints, not log spacings.
stopifnot(fit$zeta[1] < fit$zeta[2])
probability <- 1-plogis(fit$zeta[2])
rows <- list(
  list(term="speed_slope", estimate=unname(coef(fit)["speed"]), se=sqrt(V["speed","speed"]), n_obs=90L),
  list(term="high_at_zero", estimate=unname(probability), se=NULL, n_obs=90L))
