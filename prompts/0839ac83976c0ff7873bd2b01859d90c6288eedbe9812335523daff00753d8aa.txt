```python
import numpy as np
import pandas as pd
import statsmodels.api as sm
from scipy.optimize import linprog
from scipy.linalg import qr
# Smaller trade data use pair and time effects, with a zero-only pair.
rng = np.random.default_rng(835)
d = pd.DataFrame([(g,t,int(t >= 1+g%3)) for g in range(8) for t in range(5)],columns=['pair','time','agreement'])
d['trade'] = rng.poisson(np.exp(0.8 + 0.5*d.agreement + 0.06*d.pair + 0.08*d.time))
d.loc[d.pair.eq(0),'trade'] = 0
Z = np.column_stack([pd.get_dummies(d.pair,dtype=float),pd.get_dummies(d.time,dtype=float)])
X = np.column_stack([d.agreement,Z]); y=d.trade.to_numpy(); pos=y>0
separated = np.zeros(len(d), dtype=bool)
for i in np.flatnonzero(~pos):
    test = linprog(X[i], A_ub=X[~pos], b_ub=np.zeros((~pos).sum()), A_eq=X[pos], b_eq=np.zeros(pos.sum()), bounds=[(-1,1)]*X.shape[1], method='highs')
    separated[i] = test.success and test.fun < -1e-7
kept = d.loc[~separated]; Z = Z[~separated]
_,_,order = qr(Z,mode='economic',pivoting=True)
Z = Z[:,order[:np.linalg.matrix_rank(Z)]]
X = np.column_stack([kept.agreement,Z])
fit = sm.GLM(kept.trade,X,family=sm.families.Poisson()).fit()
mu=np.asarray(fit.mu); residual=kept.trade.to_numpy()-mu
bread=np.linalg.inv(X.T@(mu[:,None]*X))
scores=pd.DataFrame(X*residual[:,None]).groupby(kept.pair.to_numpy()).sum().to_numpy()
N,K=X.shape; G=kept.pair.nunique()
V=bread@scores.T@scores@bread*G/(G-1)*(N-1)/(N-K)
print({'coefficient':float(fit.params.iloc[0]), 'cluster_se':float(np.sqrt(V[0,0])), 'retained':int(N), 'ordinary_zeros':int(kept.trade.eq(0).sum())})
```
