```python
import numpy as np
import pandas as pd
from linearmodels.iv import IV2SLS
import statsmodels.api as sm
# A smaller survey estimates spillovers for two nonrecipient groups.
rng = np.random.default_rng(831)
rows = []
for eligible in [0, 1]:
    for i in range(40):
        instrument, baseline, u = rng.normal(size=3)
        dose = 2 + instrument + 0.4*u
        outcome = 1 + baseline + (0.8 + eligible)*dose + u
        rows.append([eligible, dose, instrument, baseline, outcome, 1 + i%3])
d = pd.DataFrame(rows, columns=['eligible','dose','instrument','baseline','outcome','weight'])
effects = []
slopes = []
for eligible in [0, 1]:
    h = d[d.eligible == eligible]
    fit = IV2SLS(h.outcome, sm.add_constant(h[['baseline']]), h[['dose']], h[['instrument']], weights=h.weight).fit()
    # Each group is evaluated at its own survey-weighted exposure.
    effects.append(float(fit.params['dose']*np.average(h.dose, weights=h.weight)))
    slopes.append(float(fit.params['dose']))
# Population shares differ from the equal group sizes in this survey.
population_effect = np.average(effects, weights=[60, 40])
print({'group_slopes':slopes, 'group_effects':effects, 'population_effect':float(population_effect)})
```
