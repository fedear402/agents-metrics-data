A smaller example uses separate synthetic data to illustrate the method.
```r
town <- rep(1:6,each=3); time <- rep(2:4,6)
adoption <- rep(c(3,3,0,5,4,0),each=3)
level <- rep(c(10,14,8,12,9,15),each=3)
y <- level+.5*time+2*(adoption==3 & time>=3)+c(0,.2,-.1,.1,-.2,.3,0,.1,-.1,.1,0,.2,0,.5,.8,.1,.3,-.2)
d <- data.frame(town,time,adoption,y)
d <- subset(d,!(town==6 & time==3) & (adoption==3 | adoption==0 | adoption>4))
d <- d[d$town %in% names(which(table(d$town)==3)),]
d$did <- as.numeric(d$adoption==3 & d$time>=3)
# lm includes an intercept with the omitted unit and time categories.
effect <- unname(coef(lm(y~did+factor(town)+factor(time),data=d))['did'])
changes <- sapply(split(d,d$town),function(x) mean(x$y[x$time>=3])-x$y[x$time==2])
stopifnot(abs(effect-(mean(changes[c('1','2')])-mean(changes[c('3','4')])))<1e-10)
out <- list(task_id='cohort_stack_separate_example', platform='r', estimates=list(
  list(term='town_effect',estimate=effect,se=NULL,n_obs=nrow(d)),
  list(term='treated_towns',estimate=length(unique(d$town[d$adoption==3])),se=NULL,n_obs=nrow(d))))
jsonlite::write_json(out,'result.json',auto_unbox=TRUE,null='null',digits=16)
```
